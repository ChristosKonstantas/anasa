add_executable(anasa-inference-benchmark mainInferenceBenchmark.cpp WaveWriter.hpp)
target_link_libraries(anasa-inference-benchmark PRIVATE openvpi-renderer renderer)
if(MSVC)
    target_compile_options(anasa-inference-benchmark PRIVATE /W4 /permissive-)
else()
    target_compile_options(anasa-inference-benchmark PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    file(GLOB ONNXRUNTIME_DLLS CONFIGURE_DEPENDS "${ONNXRUNTIME_ROOT}/lib/*.dll")
    if(NOT ONNXRUNTIME_DLLS)
        message(FATAL_ERROR "No ONNX Runtime DLLs found under ONNXRUNTIME_ROOT/lib")
    endif()
    add_custom_command(TARGET anasa-inference-benchmark POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ONNXRUNTIME_DLLS} "$<TARGET_FILE_DIR:anasa-inference-benchmark>"  VERBATIM)
endif()
